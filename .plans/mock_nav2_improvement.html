A. New file: my_robot_bringup/scripts/map_publisher.py

Self-contained blank map publisher. Generates a 10×10m free-space OccupancyGrid in memory. Only depends on rclpy and nav_msgs.

#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from nav_msgs.msg import OccupancyGrid


class MapPublisher(Node):
    def __init__(self):
        super().__init__("map_publisher")
        self._pub = self.create_publisher(OccupancyGrid, "map", 1,
                                          transient_local=True)
        self.create_timer(5.0, self._publish)
        self._publish()

    def _publish(self):
        msg = OccupancyGrid()
        msg.header.frame_id = "map"
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.info.resolution = 0.1
        msg.info.width = 100
        msg.info.height = 100
        msg.info.origin.position.x = -5.0
        msg.info.origin.position.y = -5.0
        msg.data = [0] * (100 * 100)
        self._pub.publish(msg)


def main():
    rclpy.init()
    node = MapPublisher()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == "__main__":
    main()

def _cmd_vel_callback(self, msg: TwistStamped):
now = self.get_clock().now()
if self._last_cmd_time is not None:
dt = (now - self._last_cmd_time).nanoseconds / 1e9
dt = min(dt, 0.1)
vx = msg.twist.linear.x
wz = msg.twist.angular.z
if abs(vx) > 0.001 or abs(wz) > 0.001:
self._x += vx * math.cos(self._yaw) * dt
self._y += vx * math.sin(self._yaw) * dt
self._yaw += wz * dt
