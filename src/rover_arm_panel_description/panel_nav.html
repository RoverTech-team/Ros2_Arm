5.2 NEW map_to_odom.py

#!/usr/bin/env python3
# map_to_odom.py
# Publishes a static map → odom transform for simulation.
# In the real system this is published by AMCL.

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import TransformStamped
from tf2_ros import StaticTransformBroadcaster

class MapToOdomPublisher(Node):
    def __init__(self):
        super().__init__("map_to_odom")
        self.declare_parameter("x", 0.0)
        self.declare_parameter("y", 0.0)
        self.declare_parameter("yaw", 0.0)

        x = self.get_parameter("x").value
        y = self.get_parameter("y").value
        yaw = self.get_parameter("yaw").value

        from tf_transformations import quaternion_from_euler
        q = quaternion_from_euler(0, 0, yaw)

        t = TransformStamped()
        t.header.frame_id = "map"
        t.child_frame_id = "odom"
        t.transform.translation.x = x
        t.transform.translation.y = y
        t.transform.translation.z = 0.0
        t.transform.rotation.x = q[0]
        t.transform.rotation.y = q[1]
        t.transform.rotation.z = q[2]
        t.transform.rotation.w = q[3]

        self._broadcaster = StaticTransformBroadcaster(self)
        self._broadcaster.sendTransform(t)
        self.get_logger().info(f"Published static map→odom: ({x}, {y}, {yaw})")

def main():
    rclpy.init()
    node = MapToOdomPublisher()
    rclpy.spin(node)
    rclpy.shutdown()

5.3 MODIFY world_to_odom.py

Complete rewrite. Subscribe to /diff_drive_controller/odom and accumulate the pose into a world→odom transform.

import rclpy
from rclpy.node import Node
from nav_msgs.msg import Odometry
from geometry_msgs.msg import TransformStamped
from tf2_ros import TransformBroadcaster

class WorldToOdom(Node):
    def __init__(self):
        super().__init__("world_to_odom")
        self._tf_broadcaster = TransformBroadcaster(self)
        self._accumulated_x = 0.0
        self._accumulated_y = 0.0
        self._accumulated_yaw = 0.0
        self._last_odom = None

        self._sub = self.create_subscription(
            Odometry, "/diff_drive_controller/odom",
            self._odom_callback, 10
        )
        self._timer = self.create_timer(0.05, self._publish_tf)

    def _odom_callback(self, msg: Odometry):
        if self._last_odom is not None:
            dx = msg.pose.pose.position.x - self._last_odom.pose.pose.position.x
            dy = msg.pose.pose.position.y - self._last_odom.pose.pose.position.y
            # Accumulate relative to starting pose
            self._accumulated_x += dx
            self._accumulated_y += dy
            # Orientation from odom quaternion
            from tf_transformations import euler_from_quaternion
            _, _, yaw = euler_from_quaternion([
                msg.pose.pose.orientation.x,
                msg.pose.pose.orientation.y,
                msg.pose.pose.orientation.z,
                msg.pose.pose.orientation.w,
            ])
            self._accumulated_yaw = yaw
        self._last_odom = msg

    def _publish_tf(self):
        if self._last_odom is None:
            return
        from tf_transformations import quaternion_from_euler
        q = quaternion_from_euler(0, 0, self._accumulated_yaw)
        t = TransformStamped()
        t.header.stamp = self.get_clock().now().to_msg()
        t.header.frame_id = "world"
        t.child_frame_id = "odom"
        t.transform.translation.x = self._accumulated_x
        t.transform.translation.y = self._accumulated_y
        t.transform.translation.z = 0.0
        t.transform.rotation.x = q[0]
        t.transform.rotation.y = q[1]
        t.transform.rotation.z = q[2]
        t.transform.rotation.w = q[3]
        self._tf_broadcaster.sendTransform(t)

Add helper function:

bool driveToPose(
    const std::string & frame_id,
    double x, double y, double yaw,
    rclcpp_action::Client<NavigateToPoseAction>::SharedPtr & client,
    rclcpp::Executor & executor,
    rclcpp::Logger & logger)
{
    if (!client->action_server_is_ready()) {
        RCLCPP_WARN(logger, "Nav2 action server not ready, skipping drive.");
        return false;
    }

    auto goal = NavigateToPoseAction::Goal();
    goal.pose.header.frame_id = frame_id;
    goal.pose.pose.position.x = x;
    goal.pose.pose.position.y = y;
    goal.pose.pose.position.z = 0.0;

    auto q = tf2::Quaternion();
    q.setRPY(0, 0, yaw);
    goal.pose.pose.orientation = tf2::toMsg(q);

    RCLCPP_INFO(logger, "Driving base to (%.2f, %.2f, %.2f)...", x, y, yaw);

    auto send_future = client->async_send_goal(goal);
    if (executor.spin_until_future_complete(send_future, std::chrono::seconds(5))
        != rclcpp::FutureReturnCode::SUCCESS) {
        RCLCPP_WARN(logger, "Failed to send Nav2 goal.");
        return false;
    }

    auto goal_handle = send_future.get();
    if (!goal_handle) {
        RCLCPP_WARN(logger, "Nav2 goal was rejected.");
        return false;
    }

    auto result_future = client->async_get_result(goal_handle);
    if (executor.spin_until_future_complete(result_future, std::chrono::seconds(60))
        != rclcpp::FutureReturnCode::SUCCESS) {
        RCLCPP_WARN(logger, "Nav2 navigation timed out.");
        return false;
    }

    auto result = result_future.get();
    if (result.code != rclcpp_action::ResultCode::SUCCEEDED) {
        RCLCPP_WARN(logger, "Nav2 navigation failed.");
        return false;
    }

    RCLCPP_INFO(logger, "Base repositioning complete.");
    return true;
}

10. Edge Cases & Risks

RiskMitigation
Nav2 cannot plan without obstacle data Blank map + allow_unknown: true ensures planner always finds a path. Only inflation layer keeps robot away from its own footprint.
map→odom static TF causes drift divergence Static TF is acceptable for short simulation runs. For longer runs, replace with an intermittent reset or simple EKF-less odom publisher.
Wheel odometry drift in mock hardware The mock hardware's calculate_dynamics: true provides idealized odometry — minimal drift.
Nav2 ignores arm-mounted panel collision object Nav2's costmap only knows the rover footprint, not the arm's extended geometry. For the demo, the arm is stowed during base movement.
Arm interacts with Nav2 costmap The arm joints are not part of Nav2's footprint. If the arm extends into obstacles during base movement, it can collide. Mitigation: Arm is stowed (upright pose) during base movement.
Timer-based Nav2 launch ordering The 18s total startup delay is conservative. Nav2 nodes may be ready earlier. The demo already waits for the arm_controller and can wait for Nav2 separately.
MacOS ROS_LOCALHOST_ONLY=1 interferes with Nav2 Nav2 runs entirely locally — all topics/actions are localhost. No issue.
